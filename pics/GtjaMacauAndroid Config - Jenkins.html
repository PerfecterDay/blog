If your job is not parameterized, then the resolved variables will just be contributed to the build. If your job is parameterized, and you resolve variables that have the same name as those parameters, then the plugin will populate the parameters when triggering job. That means you can, for example, use the parameters in combination with an SCM plugin, like GIT Plugin, to pick a branch.

Post content parameters
If you want value of param1 from post content { "param1": "value1" } to be contributed, you need to add $.param1 here.
Header parameters
If you want value of header param1 to be contributed, you need to add "param1" here.
Request parameters
If you want value of query parameter param1 to be contributed, you need to add "param1" here.
Token
Optional token. If it is specified then this job can only be triggered if that token is supplied when invoking http://JENKINS_URL/generic-webhook-trigger/invoke. It can be supplied as a:
  • Query parameter /invoke?token=TOKEN_HERE
  • A token header token: TOKEN_HERE
  • A Authorization: Bearer header Authorization: Bearer TOKEN_HERE
Token Credential
Same as token above, but configured with a secret text credential.
Cause
This will be displayed in any triggered job. You can use the variables here to create a custom cause like "$name committed to $branch", if you have configured variables named name and branch.
Allow the trigger to override this job's quiet period. If selected you can you can provide a quiet period (an integer number of seconds) and the build will use that quiet period instead of its default. If this is selected and no quiet period is given the job's quiet period will still be used. It can be supplied as a:
  • Query Parameter /invoke?jobQuietPeriod=QUIET_PERIOD_HERE
  • A quiet period header jobQuietPeriod: QUIET_PERIOD_HERE
If checked, the plugin will allow one build to have several triggers. If not checked, the plugin will trigger exactly one job when invoked.
Avoid responding with information about triggered jobs.
Avoid flattening any selected branch. If the selected node is a branch, not a leaf, the plugin will, by default, flatten its content and create variables for each leaf on that branch.
Print post content in job log.
Print contributed variables in job log.

Expression
Regular expression to test on the evaluated text specified below. The regexp syntax is documented here.
Text
Text to test for the given expression. You can use any combination of the variables you configured above.
This is an optional feature. If specified, this job will only trigger when given expression matches given text.

#!/bin/bash
# export JAVA_HOME=/opt/homebrew/opt/sdkman-cli/libexec/candidates/java/11.0.27-zulu/
# export JAVA_HOME=/opt/homebrew/opt/sdkman-cli/libexec/candidates/java/17.0.15.fx-zulu
export JAVA_HOME=/Users/gtja/.local/share/mise/installs/java/zulu-17.68.203.0/# 根据gradle.properties解析版本信息
PROPERTIES_FILE="${WORKSPACE}/gradle.properties"
versionCode_line=$(sed -n '/versionCode/p' $PROPERTIES_FILE)
versionCode=${versionCode_line#versionCode=}
versionName_line=$(sed -n '/versionName/p' $PROPERTIES_FILE)
versionName=${versionName_line#versionName=}
 
# if [ -z $VERSION_CODE ] || [ -z $VERSION_NAME ]; then
#      echo "（内部）版本号不能为空!!!"
#      exit 1
# fi
 
# 编译打包
echo "编译打包开始~"
 
channel="$(echo "$CHANNEL" | tr '[:upper:]' '[:lower:]')"
env="$(echo "$ENV" | tr '[:upper:]' '[:lower:]')"
buildType="$(echo "$BUILD_TYPE" | tr '[:upper:]' '[:lower:]')"
 
ext=apk
if [ $channel == "google" ]; then
     ext=aab
fi
 
task=assemble${CHANNEL}${ENV}${BUILD_TYPE}
finalAppFile=GTJA_JHMAC.$ext
if [ $env == "uat" ]; then
     finalAppFile=JHMAC_UAT.$ext
elif [ $env == "dev" ]; then
     finalAppFile=JHMAC_DEV.$ext
fi
headPath=app/build/outputs
midPath=apk/$channel$ENV/$buildType
if [ $channel == "google" ]; then
     task=bundle${CHANNEL}${ENV}${BUILD_TYPE}
     finalAppFile=JHMAC_$versionCode.$ext
     midPath=bundle/$channel$ENV$BUILD_TYPE
fi
 
if [ $NEED_CLEAN == 'true' ]; then
     echo "全量编译~"
     ${GRADLE_USER_HOME}/bin/gradle clean $task --stacktrace --info
else
     echo "增量编译~"
     ${GRADLE_USER_HOME}/bin/gradle $task --stacktrace --info
     if [ $? -ne 0 ]; then
         echo "增量编译失败!尝试全量编译~"
         ${GRADLE_USER_HOME}/bin/gradle clean $task --stacktrace --info
     fi
fi
 
if [ $? -ne 0 ]; then
     echo "编译打包失败!!!"
     exit 1
else
     echo "编译打包结束~"     
fi
 
# 加固和上传mapping
if [ $buildType == "release" ]; then
     # 梆梆加固(apk 56 / aab 57)
     if [ $NEED_BANGBANG_SEC == 'true' ]; then
         echo "加固开始~"
 
         originAppFile=app-$channel-$env-$buildType.$ext
         secAppFile=app-$channel-$env-$buildType-sec.$ext
         alignedAppFile=app-$channel-$env-$buildType-aligned.$ext
         strategyNum=56
         if [ $channel == "google" ]; then
             strategyNum=57
         fi
         
         java -jar ${BANGBANG_HOME}/secapi-4.1.4-SNAPSHOT.jar -i 10.180.227.155:8000 -u gtjahk -a 72b1c2d4-682a-41ff-b7a0-d1de3aec6e7e -c 88f75861-157d-46c8-a603-b8639b81f1d9 -p ${WORKSPACE}/$headPath/$midPath/$originAppFile -d ${WORKSPACE}/$headPath/$midPath/$secAppFile -t $strategyNum -f 0
         ${ANDROID_HOME}/build-tools/34.0.0/zipalign -v 4 ${WORKSPACE}/$headPath/$midPath/$secAppFile ${WORKSPACE}/$headPath/$midPath/$alignedAppFile
         ${ANDROID_HOME}/build-tools/34.0.0/apksigner sign --ks ${WORKSPACE}/app/gtjamoapp --ks-key-alias gtjamoapp --ks-pass pass:gtjamoapp123 --key-pass pass:gtjamoapp123 --min-sdk-version 24 --out ${WORKSPACE}/$headPath/$midPath/$finalAppFile ${WORKSPACE}/$headPath/$midPath/$alignedAppFile
 
         if [ $? -ne 0 ]; then
             echo "加固失败!!!"
         else
             echo "加固结束~"
         fi
     else
         finalAppFile=app-$channel-$env-$buildType.$ext
     fi
 
     # mapping上传
     if [ $NEED_UPLOAD_MAPPING == 'true' ]; then
         echo "mapping上传开始~"
 
         java -jar ${BUGLY_HOME}/buglyqq-upload-symbol.jar -appid e3305d354f -appkey 954d2ddd-bf73-4c15-88b8-6577a8f9e3c8 -bundleid com.gtjai.macau.app -version $versionName -platform Android -inputMapping ${WORKSPACE}/$headPath/mapping/$channel$ENV$BUILD_TYPE/mapping.txt
 
         if [ $? -ne 0 ]; then
             echo "mapping上传失败!!!"
         else
             echo "mapping上传结束~"
         fi
     fi
else
     finalAppFile=app-$channel-$env-$buildType.$ext
fi
 
# 写入渠道
if [ $channel == "official" ]; then
     echo "渠道写入 开始~"
     java -jar ${WALLE_HOME}/walle-cli-all.jar put -c official ${WORKSPACE}/$headPath/$midPath/$finalAppFile
     echo "渠道写入 结束~"
     fileName=$(echo $finalAppFile | cut -d . -f1)
     if [ -f "${WORKSPACE}/$headPath/$midPath/${fileName}_official.apk" ]; then
         rm ${WORKSPACE}/$headPath/$midPath/$finalAppFile
         mv ${WORKSPACE}/$headPath/$midPath/${fileName}_official.apk ${WORKSPACE}/$headPath/$midPath/$finalAppFile
     fi
     echo "写入的渠道是："
     java -jar ${WALLE_HOME}/walle-cli-all.jar show ${WORKSPACE}/$headPath/$midPath/$finalAppFile
fi
 
# Set build desc
# 生成二维码和下载链接(内网)
if [ -f "${WORKSPACE}/$headPath/$midPath/$finalAppFile" ]; then
     echo "生成二维码 开始~"
     
     local_ip=`ifconfig en1|grep inet|grep -v 127.0.0.1|grep -v inet6|awk '{print $2}'|tr -d "addr:"`
     url_prefix="http://${local_ip}:8080/job/GtjaMacauAndroid/${BUILD_NUMBER}/artifact"
     des="[DESC]<img src=\"$url_prefix/$headPath/$midPath/qrcode.png\" width=\"100\" height=\"100\"><br><a href=\"$url_prefix/$headPath/$midPath/$finalAppFile\">下载连接</a>"
 
     python3 /Users/gtja/Applications/amazing-qr/amzqr.py $url_prefix/$headPath/$midPath/$finalAppFile -v 7 -l L -d ${WORKSPACE}/$headPath/$midPath
     if [ $? -ne 0 ]; then
         echo "生成二维码 失败!!!"
         des="[DESC]<a href=\"$url_prefix/$headPath/$midPath/$finalAppFile\">下载连接</a>"
     else
         echo "生成二维码 结束~"
     fi
     echo "${des}"
fi
 
# 独立于Jenkins自带的归档功能，为了方便根据版本查看
echo "归档开始~"
 
archivedDir=${JENKINS_HOME}/JHQQT_MACAU_ARCHIVED/$versionName/${BUILD_DISPLAY_NAME}
if [ ! -d "$archivedDir" ]; then
     # if body
     mkdir -p $archivedDir
fi
 
appToCopy=${WORKSPACE}/$headPath/$midPath/$finalAppFile
if [ -f "$appToCopy" ]; then
     cp -r $appToCopy $archivedDir/$finalAppFile
else
     echo "$appToCopy 不存在!"
fi
 
qrcodeToCopy=${WORKSPACE}/$headPath/$midPath/qrcode.png
if [ -f "$qrcodeToCopy" ]; then
     cp -r $qrcodeToCopy $archivedDir/qrcode.png
else
     echo "$qrcodeToCopy 不存在!"
fi
 
if [ $buildType == "release" ]; then
     confToCopy=${WORKSPACE}/$headPath/mapping/$channel$ENV$BUILD_TYPE/configuration.txt
     if [ -f "$confToCopy" ]; then
         cp -r $confToCopy $archivedDir/configuration.txt
     else
         echo "$confToCopy 不存在!"
     fi
 
     mappingToCopy=${WORKSPACE}/$headPath/mapping/$channel$ENV$BUILD_TYPE/mapping.txt
     if [ -f "$mappingToCopy" ]; then
         cp -r $mappingToCopy $archivedDir/mapping.txt
     else
         echo "$mappingToCopy 不存在!"
     fi
 
     seedsToCopy=${WORKSPACE}/$headPath/mapping/$channel$ENV$BUILD_TYPE/seeds.txt
     if [ -f "$seedsToCopy" ]; then
         cp -r $seedsToCopy $archivedDir/seeds.txt
     else
         echo "$seedsToCopy 不存在!"
     fi
fi
 
echo "归档结束~"
